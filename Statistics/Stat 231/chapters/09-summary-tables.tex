\section{Likelihood-based results}

\begin{itemize}
  \item Relative likelihood $R(\theta) = L(\theta)/L(\thetahat)$; $100p\%$ likelihood
        interval $\{\theta: R(\theta)\ge p\}$.
  \item $\Lambda(\theta) = -2\log R(\theta)\approx\chi^2(1)$; $15\%$ LI $\approx$ $95\%$ CI,
        $10\%$ LI $\approx$ $96.8\%$ CI, $1\%$ LI $\approx$ $99.8\%$ CI.
  \item LRT of $H_0:\theta=\theta_0$: $p\approx\Prob(W\ge\lambda) =
        2[1-\Phi(\sqrt\lambda)]$, $W\sim\chi^2(1)$.
  \item Goodness of fit: $\Lambda = 2\sum y_j\log(y_j/e_j)$, $D = \sum(y_j-e_j)^2/e_j$,
        $\chi^2(k-1-p)$; independence: $e_{ij} = r_ic_j/n$, $\chi^2((a-1)(b-1))$.
\end{itemize}

\section{Workflow reminders}

\begin{enumerate}
  \item Problem and Plan: identify target and study populations, variates, sampling
        protocol; anticipate study, sample and measurement error.
  \item Plot the data; choose a model; estimate by maximum likelihood; check the fit.
  \item Report estimates with confidence or likelihood intervals; test hypotheses with
        p-values interpreted on the evidence scale.
  \item Draw conclusions in context, distinguishing association from causation.
\end{enumerate}
